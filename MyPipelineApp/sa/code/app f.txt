

def generate_cpp_from_template_safe_replace(blueprint_data, cpp_template_string):
    if not blueprint_data:
        return "// Blueprint data not provided."

    # --- Extract data from blueprint for replacements ---
    project_details = blueprint_data.get("project_details", {})
    project_name = project_details.get("name", "Untitled Project")
    
    esp32_settings = blueprint_data.get("esp32_settings", {})
    target_mode_str = esp32_settings.get("target_mode", "REAL_SENSORS")
    
    tflm_options = blueprint_data.get("tflm_options", {})
    tflm_enabled_bool = tflm_options.get("enable_tflm", False)
    
    network_cfg = blueprint_data.get("network_info", {})
    wifi_ssid = network_cfg.get("ssid", "DEFAULT_SSID")
    wifi_password = network_cfg.get("password", "DEFAULT_PASSWORD")

    pipe_props = blueprint_data.get("pipe_physical_properties", {})
    pipeline_length = pipe_props.get("mainline_length_m", 0.9)
    pipe_diameter = pipe_props.get("mainline_inner_diameter_m", 0.006)
    pipe_roughness = pipe_props.get("mainline_pipe_roughness_m", 0.0000015)

    system_timing_cfg = blueprint_data.get("system_timing", {})
    update_interval_ms = system_timing_cfg.get("update_interval_ms", 1000)

    main_sensors_cfg = blueprint_data.get("main_flow_sensors", {})
    num_main_sensors_val = main_sensors_cfg.get("count", 0) 
    global_k_val = main_sensors_cfg.get("global_k_factor_pulses_per_liter", 450.0)

    minor_loss_cfg = blueprint_data.get("minor_loss_coefficients", {})
    tuning_cfg = blueprint_data.get("system_tuning_parameters", {})

    # --- Prepare values for substitution ---
    cpp_comment_header_val = f"""//========================================================================== //
// ESP32 Leak Detection System - Generated Code
// Generated by Python Configurator App on: {datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")}
// Project: {project_name}
// Blueprint Target Mode: {target_mode_str}
// TFLM Enabled in Blueprint: {"Yes" if tflm_enabled_bool else "No"}
// Number of Main Sensors from Blueprint: {num_main_sensors_val}
//========================================================================== //"""

    use_tflm_define_val = "1" if tflm_enabled_bool else "0"
    simulate_real_sensors_define_val = "true" if target_mode_str == "REAL_SENSORS" else "false"
    
    # Start with the C++ template that has placeholders
    generated_code = cpp_template_string

    # Perform .replace() for each placeholder
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_CPP_COMMENT_HEADER__", cpp_comment_header_val)
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_USE_TFLM_DEFINE__", use_tflm_define_val)
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_SIMULATE_REAL_SENSORS_DEFINE__", simulate_real_sensors_define_val)
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_NUM_MAIN_SENSORS__", str(num_main_sensors_val))
    
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_GLOBAL_K_FACTOR__", f"{global_k_val:.1f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_PIPELINE_LENGTH__", f"{pipeline_length:.2f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_PIPE_DIAMETER__", f"{pipe_diameter:.4f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_PIPE_ROUGHNESS_M__", f"{pipe_roughness:.7f}")
    
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_K_90_ELBOW__", f"{minor_loss_cfg.get('k_90_elbow', 0.9):.1f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_K_45_ELBOW__", f"{minor_loss_cfg.get('k_45_elbow', 0.4):.1f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_K_TEE_BRANCH__", f"{minor_loss_cfg.get('k_tee_branch', 1.8):.1f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_K_TEE_STRAIGHT__", f"{minor_loss_cfg.get('k_tee_straight', 0.6):.1f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_NUM_90_ELBOWS__", str(minor_loss_cfg.get('num_90_elbows', 0)))
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_NUM_45_ELBOWS__", str(minor_loss_cfg.get('num_45_elbows', 0)))
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_NUM_TEE_BRANCHES__", str(minor_loss_cfg.get('num_tee_branches', 0)))
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_NUM_TEE_STRAIGHT__", str(minor_loss_cfg.get('num_tee_straight', 0)))

    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_WIFI_SSID__", f"\"{wifi_ssid}\"")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_WIFI_PASSWORD__", f"\"{wifi_password}\"")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_UPDATE_INTERVAL_MS__", str(update_interval_ms))

    # Tuning Parameters
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_EMA_ALPHA__", f"{tuning_cfg.get('ema_alpha', 0.4):.2f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_CALIBRATION_SAMPLE_COUNT__", str(tuning_cfg.get('calibration_sample_count', 10)))
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_DEVIATION_STD_DEV_FACTOR__", f"{tuning_cfg.get('deviation_std_dev_factor', 1.0):.1f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_CALIBRATION_MIN_START_FLOW_LPS__", f"{tuning_cfg.get('calibration_min_start_flow_lps', 0.1):.3f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_NEAR_ZERO_FLOW_LPS__", f"{tuning_cfg.get('near_zero_flow_lps', 0.05):.3f}")
    
    max_invalid_ratio = tuning_cfg.get('max_invalid_samples_during_calib_ratio', 0.333)
    calib_sample_count = tuning_cfg.get('calibration_sample_count', 10)
    max_invalid_abs = int(calib_sample_count * max_invalid_ratio)
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_MAX_INVALID_SAMPLES_DURING_CALIB__", str(max_invalid_abs))
    
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_CALIB_START_CONFIRM_CYCLES__", str(tuning_cfg.get('calib_start_confirm_cycles', 3)))
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_HYSTERESIS_FACTOR__", f"{tuning_cfg.get('hysteresis_factor', 0.8):.2f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_LEAK_CONFIRMATION_CYCLES_DEFAULT__", str(tuning_cfg.get('leak_confirmation_cycles_default', 3)))
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_ADAPTIVE_CONFIRM_CYCLES_FAST__", str(tuning_cfg.get('adaptive_confirm_cycles_fast', 1)))
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_ADAPTIVE_CONFIRM_FACTOR_HIGH__", f"{tuning_cfg.get('adaptive_confirm_factor_high', 3.0):.1f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_DRIFT_CONFIRMATION_CYCLES__", str(tuning_cfg.get('drift_confirmation_cycles', 5)))
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_BURST_ROC_THRESHOLD_LPS_SEC__", f"{tuning_cfg.get('burst_roc_threshold_lps_sec', 0.05):.2f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_GENERAL_ROC_WARN_THRESHOLD_LPS_SEC__", f"{tuning_cfg.get('general_roc_warn_threshold_lps_sec', 0.15):.2f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_BASELINE_DRIFT_FACTOR__", f"{tuning_cfg.get('baseline_drift_factor', 4.0):.1f}")

    near_zero_flow_lps_val = tuning_cfg.get('near_zero_flow_lps', 0.050)
    near_zero_diff_ratio = tuning_cfg.get('near_zero_diff_lps_ratio_to_near_zero_flow', 0.2)
    near_zero_diff_lps_val = near_zero_flow_lps_val * near_zero_diff_ratio
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_NEAR_ZERO_DIFF_LPS__", f"{near_zero_diff_lps_val:.3f}")
    
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_MAX_REALISTIC_FLOW_LPS__", f"{tuning_cfg.get('max_realistic_flow_lps', 20.0):.1f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_MAX_CONSECUTIVE_FAULTS__", str(tuning_cfg.get('max_consecutive_faults', 10)))
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_STUCK_SENSOR_CYCLES__", str(tuning_cfg.get('stuck_sensor_cycles', 10)))
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_VALIDITY_MAX_RAW_ROC_LPS__", f"{tuning_cfg.get('validity_max_raw_roc_lps', 15.0):.1f}")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_DEBUG_VALIDITY_LEVEL__", str(tuning_cfg.get('debug_validity_level', 1)))
    
    min_delta_loc_ratio = tuning_cfg.get('min_total_pos_delta_for_location_estimate_ratio_to_near_zero_flow', 0.1)
    min_delta_loc_val = near_zero_flow_lps_val * min_delta_loc_ratio 
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_MIN_TOTAL_POS_DELTA_FOR_LOCATION_ESTIMATE__", f"{min_delta_loc_val:.3f}")
    
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_MAX_LOG_EVENTS__", str(tuning_cfg.get('max_log_events', 15)))

    # Placeholder comment sections (these will be filled by other Python functions in later steps)
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_MAIN_SENSOR_PIN_DEFINES__", "// [INFO: Main sensor pin definitions will be dynamically generated here (Step 1.D)]")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_PER_SENSOR_K_FACTOR_DEFINES__", "// [INFO: Per-sensor K-factor definitions will be dynamically generated here if enabled (Step 1.E)]")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_GLOBAL_N_SENSOR_VARIABLES__", "// [INFO: Global variables for N sensors will be dynamically generated here (Step 1.F)]")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_ISR_DEFINITIONS__", "// [INFO: ISR functions for N sensors will be dynamically generated here (Step 1.G)]")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_SETUP_PIN_INIT_AND_ISR_ATTACH__", "// [INFO: pinMode and attachInterrupt calls for N sensors will be generated here (Step 1.G)]")
    generated_code = generated_code.replace("__PYTHON_PLACEHOLDER_PROJECT_NAME_IN_SETUP__", project_name) # For the Serial.println in setup()

    return generated_code

# --- Main execution block ---
